\section{Wymagania funkcjonalne}

\begin{itemize} 
	\item Użytkownik ma mieć możliwość rejestracji i logowania się do systemu
	\item Aplikacja ma obsługiwać role gościa, użytkownika i administratora
	\item Administrator ma mieć możliwość zarządzania użytkownikami (dodawanie, usuwanie, edytowanie)
	\item Użytkownik ma mieć możliwość kupowania żetonów zwykłych oraz specjalnych żetonów premium
	\item Aplikacja ma zawierać gry:
		\begin{itemize}
			\item Ruletka
			\item Sloty
			\item Zgadywanie liczby
		\end{itemize}
	\item W grach, żetony premium powinny zwiększyć szanse na wygraną
	\item Użytkownicy powinni móc dodawać awatary do swoich profili
	\item Powinien istnieć ranking użytkowników, działający na podstawie posiadanych żetonów
\end{itemize}

\section{Wymagania niefunkcjonalne}

\begin{itemize}
	\item Aplikacja ma być dostępna przez przeglądarkę internetową
	\item Aplikacja będzie się składać w większości z statycznych stron, renderowanych po stronie serwera
	\item Gry będą wykorzystywały JavaScript do interakcji z użytkownikiem i animacji 
	\item Serwer będzie używał prostego API do komunikacji z klientem podczas grania
	\item Aplikacja ma komunikować użytkownikowi błędy w czytelny sposób
\end{itemize}
